import pandas as pd
import os

# Step 1: Set the path to the folder containing the files
extracted_folder = r"File path"

# Step 2: List all files in the folder
files = os.listdir(extracted_folder)

# Step 3: Initialize an empty list to store DataFrames
dfs = []

# Step 4: Load data from CSV files and append to the list
for file in files:
    if file.lower().endswith('.csv'):  # Check for .csv extensions (case-insensitive)
        file_path = os.path.join(extracted_folder, file)
        try:
            # Read the CSV file using the second row as the header (skip the first row)
            df = pd.read_csv(file_path, skiprows=1)  # Skip the first row and use row 2 as headers
            dfs.append(df)
            print(f"Successfully loaded: {file}")
        except Exception as e:
            print(f"Error reading {file}: {e}")

# Step 5: Combine all loaded DataFrames into one
if dfs:  # Check if there are any DataFrames
    full_data = pd.concat(dfs, ignore_index=True)
    print("Data combined successfully!")
else:
    print("No CSV files found or successfully loaded.")
    full_data = pd.DataFrame()  # Ensure full_data exists for further processing

# Step 6: Filter REGIONID for valid values
valid_regions = ['NSW1', 'QLD1', 'VIC1', 'SA1', 'TAS1']  # Adjust based on valid regions
full_data = full_data[full_data['REGIONID'].isin(valid_regions)]

# Step 7: Convert INTERVAL_DATETIME to datetime format and split into Date and Time columns
full_data['INTERVAL_DATETIME'] = pd.to_datetime(full_data['INTERVAL_DATETIME'], errors='coerce')
full_data['Interval Date'] = full_data['INTERVAL_DATETIME'].dt.date
full_data['Interval Time'] = full_data['INTERVAL_DATETIME'].dt.time

# Step 8: Handle OPERATIONAL_DEMAND and OPERATIONAL_DEMAND.1
if 'OPERATIONAL_DEMAND.1' in full_data.columns:
    # Replace OPERATIONAL_DEMAND with OPERATIONAL_DEMAND.1 values if OPERATIONAL_DEMAND is invalid
    full_data['OPERATIONAL_DEMAND'] = full_data['OPERATIONAL_DEMAND.1']
    full_data.drop(columns=['OPERATIONAL_DEMAND.1'], inplace=True)

# Step 9: Remove unnecessary columns (e.g., WDR_ESTIMATE)
columns_to_drop = ['WDR_ESTIMATE']
full_data.drop(columns=columns_to_drop, inplace=True)

# Step 10: Rename the column '3' to 'AEMO'
full_data.rename(columns={'3': 'AEMO'}, inplace=True)

# Step 11: Check data integrity and display cleaned data
print("\nFirst three rows of the modified data:")
print(full_data.head(3))

print("\nUpdated Data Information:")
print(full_data.info())

print("\nUpdated Summary Statistics:")
print(full_data.describe(include='all'))

# Step 12: Export the cleaned data to a CSV file
output_file = r"C:\Users\Maiwa\OneDrive\Desktop\Evoenergy_Demand_Cleaned_Data.csv"
full_data.to_csv(output_file, index=False)
print(f"Cleaned data successfully saved to {output_file}.")

import pandas as pd
from statsmodels.tsa.arima.model import ARIMA
import matplotlib.pyplot as plt

# Load the cleaned data
data_file = r"C:\Users\Maiwa\OneDrive\Desktop\Evoenergy_Demand_Cleaned_Data.csv"
data = pd.read_csv(data_file)

# Convert 'Interval Date' to datetime and aggregate daily demand
data['Interval Date'] = pd.to_datetime(data['Interval Date'])
daily_demand = data.groupby('Interval Date')['OPERATIONAL_DEMAND'].sum()
daily_demand.index.freq = 'D'  # Explicitly set frequency to daily

# Fit ARIMA model
try:
    # Order (p, d, q) values can be adjusted based on the dataset
    arima_model = ARIMA(daily_demand, order=(1, 1, 1))
    arima_result = arima_model.fit()

    # Forecast the next 31 days
    forecast = arima_result.forecast(steps=31)

    # Print forecast results
    print("Forecasted Energy Demand (MW) for the next 31 days:")
    forecast_dates = pd.date_range(start=daily_demand.index.max() + pd.Timedelta(days=1), periods=31, freq='D')
    forecast_table = pd.DataFrame({'Date': forecast_dates, 'Forecasted Demand (MW)': forecast})
    print(forecast_table)

    # Plot historical and forecasted data
    plt.figure(figsize=(10, 6))
    plt.plot(daily_demand, label='Historical Demand')
    plt.plot(forecast_dates, forecast, label='Forecast', linestyle='--', color='red')
    plt.title('Energy Demand Forecast (ARIMA)')
    plt.xlabel('Date')
    plt.ylabel('Energy Demand (MW)')
    plt.legend()
    plt.grid(True)
    plt.show()

except Exception as e:
    print(f"Error fitting ARIMA model: {e}")
